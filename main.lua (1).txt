    print("Asset ID:", assetId)
    
    local methods = {
        function()
            print("🔄 [1/6] Direct Require")
            return pcall(function() require(tonumber(assetId)) end)
        end,
        
        function()
            print("🔄 [2/6] GetObjects")
            local s, r = pcall(function()
                local objs = game:GetObjects("rbxassetid://" .. assetId)
                for _, obj in pairs(objs) do
                    if obj:IsA("ModuleScript") then
                        require(obj)
                        return true
                    end
                end
                return false
            end)
            return s and r
        end,
        
        function()
            print("🔄 [3/6] InsertService")
            local s, r = pcall(function()
                local model = game:GetService("InsertService"):LoadAsset(tonumber(assetId))
                if model then
                    for _, obj in pairs(model:GetDescendants()) do
                        if obj:IsA("ModuleScript") then
                            require(obj)
                            model:Destroy()
                            return true
                        end
                    end
                end
                return false
            end)
            return s and r
        end,
        
        function()
            print("🔄 [4/6] ContentProvider")
            local s, r = pcall(function()
                local asset = "rbxassetid://" .. assetId
                game:GetService("ContentProvider"):PreloadAsync({asset})
                local objs = game:GetObjects(asset)
                for _, obj in pairs(objs) do
                    if obj:IsA("ModuleScript") then
                        require(obj)
                        return true
                    end
                end
                return false
            end)
            return s and r
        end,
        
        function()
            print("🔄 [5/6] Clone Method")
            local s, r = pcall(function()
                local objs = game:GetObjects("rbxassetid://" .. assetId)
                for _, obj in pairs(objs) do
                    if obj:IsA("ModuleScript") then
                        local clone = obj:Clone()
                        clone.Parent = game:GetService("ReplicatedStorage")
                        require(clone)
                        task.wait(0.05)
                        clone:Destroy()
                        return true
                    end
                end
                return false
            end)
            return s and r
        end,
        
        function()
            print("🔄 [6/6] HTTP LoadString")
            local urls = {
                "https://raw.githubusercontent.com/scripts/" .. assetId,
                "https://pastebin.com/raw/" .. assetId,
            }
            for _, url in ipairs(urls) do
                local s, code = pcall(function() return game:HttpGet(url) end)
                if s and code and #code > 10 then
                    pcall(function() loadstring(code)() end)
                    return true
                end
            end
            return false
        end,
    }
    
    local success = 0
    for i, method in ipairs(methods) do
        if method() then
            print(" Require Method " .. i .. " SUCCESS")
            success = success + 1
        else
            print(" Require Method " .. i .. " FAILED")
        end
        task.wait(0.08)
    end
    
    print("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
