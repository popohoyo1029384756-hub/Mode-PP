-- [[ ระบบลาก (Smooth Dragging) ]]

-- ==========================================

local UIS = game:GetService("UserInputService")

local dragFrame = LMG2L["Frame_2"]

local dragHandle = LMG2L["Frame_8"]

local dragToggle, dragStart, startPos

dragHandle.InputBegan:Connect(function(input)

	if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then		dragToggle = true

		dragStart = input.Position

		startPos = dragFrame.Position

		input.Changed:Connect(function()

			if input.UserInputState == Enum.UserInputState.End then dragToggle = false end

		end)

	end

end)

UIS.InputChanged:Connect(function(input)

	if dragToggle and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then

		local delta = input.Position - dragStart

		game:GetService("TweenService"):Create(dragFrame, TweenInfo.new(0.1), {

			Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)

		}):Play()

	end

end)

-- [[ แก้ไข: ขอบสีฟ้าที่ "ขอบปุ่ม" ไม่ใช่รอบตัวหนังสือ ]]

-- ==========================================

local function ApplyBlueBorder(button)

	local stroke = Instance.new("UIStroke")

	stroke.Color = Color3.fromRGB(0, 170, 255)

	stroke.Thickness = 3

	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border -- บังคับให้เป็นขอบของวัตถุ ไม่ใช่ตัวหนังสือ

	stroke.Enabled = false

	stroke.Parent = button

	button.MouseButton1Down:Connect(function()

		stroke.Enabled = true

	end)

	button.InputEnded:Connect(function(input)

		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

			stroke.Enabled = false

		end

	end)

end

-- นำไปใช้กับทุกปุ่ม ยกเว้น TextBox

for _, obj in pairs(LMG2L) do

	if obj:IsA("TextButton") then

		ApplyBlueBorder(obj)

	end

end

LMG2L["TextButton3_a"].MouseButton1Click:Connect(function()

	local player = game.Players.LocalPlayer

	local character = player.Character

	

	if character and character:FindFirstChild("HumanoidRootPart") then

		local oldPos = character.HumanoidRootPart.CFrame -- เก็บพิกัดเดิมไว้

		

		player:LoadCharacter() -- สั่งเกิดใหม่ทันที

		

		-- รอตัวละครใหม่เข้าเกมและวาร์ปไปจุดเดิม

		task.spawn(function()

			local newChar = player.CharacterAdded:Wait()

			local hrp = newChar:WaitForChild("HumanoidRootPart", 5)

			if hrp then

				task.wait(0.1) -- รอจังหวะนิดเดียวเพื่อให้ระบบโหลดพิกัดเสร็จ

				hrp.CFrame = oldPos

				LMG2L["TextBox_e"].Text = "RE Success: Respawned at Sport!"

				LMG2L["TextBox_e"].TextColor3 = Color3.fromRGB(0, 170, 0)

			end

		end)

	end

end)

-- [[ EXE Button ]]

LMG2L["TextButton_4"].MouseButton1Click:Connect(function()

	local code = LMG2L["input_7"].Text

	local output = LMG2L["TextBox_e"]

	if code == "" or code:match("^%s*$") then

		output.Text = "Error: No script provided!"

		output.TextColor3 = Color3.fromRGB(255, 0, 0)

		return

	end

	local success, func = pcall(loadstring, code)

	if success then

		pcall(func)

		output.Text = "Success: Executed!"

		output.TextColor3 = Color3.fromRGB(0, 170, 0)

	else

		output.Text = "Error: " .. tostring(func)

		output.TextColor3 = Color3.fromRGB(255, 0, 0)

	end

end)
